## Ansible - setup Nginx with reverse Proxy

'''yaml
- name: Setup Nginx with reverse Proxy
  hosts: all
  become: true
  tasks:
    - name: install Nginx
      yum:
        name: nginx
        state: present
    - name: Configure nginx as reverse Proxy
      templete:
        src: reverse-proxy.conf.j2
        dest: /etc/nginx.conf.d/reverse-proxy.conf
    - name: Restart nginx service
      service:
        name: nginx
        state: restarted
        enabled: yes

#Kubernetes - Ingress with nginx controller

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: pathnex-Ingress
  annotations:
    nginx.ingress.kubernetes.io/rewrite-targets: /
spec:
  rules:
    - host: pathnex.local
      http:
        paths:
          - path: /
            pathType: prefix
            backend:
              service:
                name: pathnex-service
                port:
                  number: 80

#Jenkinsfile - Build, test. deploy and stages

    agent any

        stage ('Build') {
            steps {

        stage('Test') {

                }
            }

    }

#Gitlab CI/CD - Multiple stage Deployment with helm

stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t pathnex-web-app .

push:
  stage: push
  script:
    script:
      - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
      - docker push pathnex-web-app

deploy:
  stage: deploy
  script:
    - helm upgrade --install pathnex-nginx pathnex/nginx-ingress
