Day - 04 Basic CI/CD Concept with Jenkins and Gitlab

Jenkinsfile - Basic CI pipeline
pipeline {
    agent any
    stages {
        stage('build') {
            steps {
                echo 'Building Project'
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying Application'
            }
        }
    }
}

#Gitlab CI/CD - basic pipeline

stages:
  - build
  - test
  - deploy

build:
  stage: build
  script:
    - echo "Building Project"

test:
  stage: test
  script:
    - echo "Running tests"

deploy:
  stage: deploy
  script:
    - echo "Deploying Application"
    
#Ansible - Install and configure Nginx
- name: Install and configure Nginx
  hosts: all
  become: true
  tasks:
    - name: Install Nginx
      yum:
        name: nginx
        state: present
    
    - name: configre nginx
      templete:
        src: nginx.conf.j2
        dest: /etc/nginx/nginx.conf
    
    - name: start nginx service
      service:
        name: nginx
        state: started
        enabled: true
