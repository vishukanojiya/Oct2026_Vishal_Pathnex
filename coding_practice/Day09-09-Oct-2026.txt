#Ansible - setup Docker container for nginx

-name: setup Docker container for nginx
hosts: all
become: yes
tasks:
  - name: pull nginx image
    docker_image:
      name: nginx
      source: pull
  - name : Run nginx container
    docker_container:
      name: nginx-container
      image: nginx
      state: startedd
      punlished_port
        - "8080:80"

#Kubernetes - Horizontal pod autoscaler (HPA)
apiVersion: app/v1
kind: deployment
metadata: 
  name: pathnex-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: pathnex-app
    templete:
      metadata:
        labels:
          app: pathnex-app
      spec:
        containers:
          - name: nginx
            image: nginx
            port:
              - containerPort: 80

---

apiVersion: app/v1
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-hpa
spec:
  scaleTargetRef:
    apiVersion: app/v1
    kind: deployment
    name: pathnex-deployment
  minReplicas: 1
  maxReplicas: 10
  targetCPUUtilicationPercentage: 50
