Day 03 Ansible, Terraform & kubernetes

Ansible - Install apache and start service

- name: Install and start apache
  hosts: all
  become: yes
  tasks:
    - name: apache Install
        yum:
          name: httpd
          state: present

    - name: start service
      service:
        name: httpd
        state: started
        enabled: yes

#kubernetes - Create a replicaSet with 3 replicaSet

kind: replicaSet
metadata:
  name: pathnex-replicaSet
spec:
  replicas: 3
  selector:
    matchLabels:
      app: pathnex-app
    templete:
      metadata:
        labels:
          app: pathnex-app
      spec:
        containers:
          - name: app
            image: nginx
            ports:
              - containerPort: 80
