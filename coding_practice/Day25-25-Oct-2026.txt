#Terraform - Create Cloudwatch Alerts

resource "aws_cloudwatch-matric_alarm" "pathnex_cpu_alerm" {
    alerm_name = "pathnex_cpu_alerm_high"
    comparison_operator = "GreaterThanThreshold"
    evaluation_periods = "1"
    matric_name = "CPUUtilization"
    namespace = "AWS/EC2"
    period = "60"
    statistic = "Average"
    threshold = "80"

    dimensions = {
        InstanceId = aws_instance.pathnex.ex2.id
    }
    alerm_action = [aws_sns_topic.pathnex_alerts.arn]
}

resource "aws_sns_topic" "pathnex_alerts" {
    name = "pathnex_alerts"
}

#JenkinsFile - COntinuous monitoring and alerts in CI pipeline
pipeline {
    agent any
    stages {
        stage('Monitor') {
            steps {
                script {
                    echo 'Triggering CloudWatch alerms if CPU exceeds 80%'
                    sh 'aws CloudWatch put-matric-data --namespace "AWS/EC2" matric-name "CPUUtilization" --value 90'
                }
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests'
            }
        }
    }
}
