#Ansible - Deploying Nginx across multiple Kubernetes clusters

- name: Dploying nginx across multiple kubernetes clusters
  hosts: localhost
  tasks:
    - name: configure kubectl for cluster 1
      command: kubectl --context=us-east config use-context us-east-cluster
    - name: Deploy nignx on cluster 1
      command: kubectl apply -f kubernetes/deploment.yaml
    - name: configure kubectl for cluster 2
      command: kubectl --context=us-west config use-context us-west-cluster
    - name: configure kubeclt cluster 2
      command: kubectl apply -f kubernetes/deploment.yaml

#JenkinsFile - multi-Cluster Deployment strategy
pipeline {
    agent any
    stages {
        stage('Deploy cluster 1') {
            steps {
                script {
                    sh 'kubectl --context=us-east-cluster apply -f kubernetes/deployment.yaml'
                }
            }
        }
        stage(Deploy cluster 2) {
            steps {
                script {
                    sh 'kubectl --context=us-east-cluster apply -f kubernetes/deployment.yaml'
                }
            }
        }
    }
}
