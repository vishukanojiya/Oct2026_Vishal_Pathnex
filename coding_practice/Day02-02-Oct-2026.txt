Day 02 - Services & Tags

Ansible - Install & Enable Nginx
- name: Install and start Nginx on Pathnex
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present

    - name: Enable nginx
      services:
        name: nginx
        state: started
        enabled: yes

# Terraform - EC2 with tags (r5.2xlarge)

#Kubernetes - Deployment with 2 Replicas

apiVersion: app/v1
kind: Deployment
metaData:
  app: Pathnex-Deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: pathnex-app
    
    templete:
      metaData:
        labels:
          app: pathnex-app
      spec:
        containers:
          - name: app
            image: nginx
            port:
              - containerPort: 80
