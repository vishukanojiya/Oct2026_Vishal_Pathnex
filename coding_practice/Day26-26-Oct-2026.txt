#Day 26 - Security Best Practices

#Ansible - Implementing SSH key-based Authentication
- name: Implement SSH key Authentication
  hosts: all
  become: yes
  tasks:
    - name: Generate SSH key Pair
      command: ssh-keygen -t rsa -b 2048 -f /home/user/.ssh/id_rsa -N
    - name: Add SSH public key to authorized_keys
      authorized_keys:
        user: user
        state: present
        key: "{{ lookup('file', '/home/user/.ssh/id_rsa.pub') }}"


#Terraform - Enabling VPC flow logs for Security
resource "aws_vpc_flow_log" "pathnex_flow_log" {
    log_group_name = "pathnex_flow_log"
    vpc_id         = aws_vpc.pathnex_vpc.id
    traffic_type   = "ALL"
    log_destination_type = "cloud-watch-log"
}

resource "aws_cloudwatch_log_group" "pathnex_flow_log_group" {
    name = "pathnex-flow-log"
}

#Kubernetes - Network policies for Security
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: pathnex-web-policy
spec:
  podSelector:
    matchLabels:
      app: pathnex-web-app
  policyTypes:
    - ingress
    - Egress
  ingress:
    - from:
      - podSelector:
          matchLabels:
            app: pathnex-api
  egress:
    - to
      - podSelector:
          matchLabels:
            app: pathnex-database

#Jenkinsfile - Secure the application with Secrets

pipeline {
    agent any
    environment {
        AWS_ACCESS_KEY_ID = credentials('aws-access-key')
        aws_SECRET_ACCESS_KEY = credentials('aws-secret-key')
    }
    stages {
        stage('Build') {
            steps {
                echo 'building docker images'
                docker.build('pathnex-web-app')
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying to AWS EC2'
                sh 'aws ec2' run-instances --image-id ami-0abcd1234 --count 1 --instance-type t2.micro
            }
        }
    }
}


#GitLab - Use GitLab Secrets for Secure Deployment
stages:
  - build
  - Deploy

  build:
    stage: build
    script:
      - docker build -t pathnex-web-app .

  deploy:
    stage: deploy
    script:
      - echo "Deploying with secure credentials"
      - aws ec2 run-instances --image-id ami-0abcd1234 --count 1 --instance-type t2.micro --access-key $AWS_ACCESS_KEY --secret-key $AWS_SECRET_KEY
